## Phase 14: Attention Interface

**Commit**: `refactor: create attention interface with swappable implementations`

**Prerequisite**: All previous phases complete. `tinygpt/attention.py` exists with a single `CausalSelfAttention` class.

**Create**: `tinygpt/attention.py` (replace contents)

### Interface

```python
from abc import ABC, abstractmethod

class Attention(ABC):
    """Base class for all attention implementations."""

    @abstractmethod
    def __init__(self, embed_dim: int, num_heads: int, dropout: float):
        pass

    @abstractmethod
    def forward(self, x) -> torch.Tensor:
        """
        Args:
            x: [B, T, C] input tensor
        Returns:
            [B, T, C] output tensor
        """
        pass

    @abstractmethod
    def extra_repr(self) -> str:
        """String representation for debugging."""
        pass
```

### Implementations

**1. SoftmaxAttention** (standard O(N²))
- Manual `q @ k.T` scaling, causal mask, softmax, dropout
- For comparison/baseline

**2. FlashAttention** (O(N) memory)
- `F.scaled_dot_product_attention(q, k, v, is_causal=True)`
- Current default — fastest on CUDA

**3. LinearAttention** (O(N) compute)
- Kernel trick: `φ(q) @ φ(k).T @ v`
- Approximate, faster for long sequences
- No causal mask by default (additive mask variant)

**4. HybridAttention** (chunked)
- Split sequence into chunks of `chunk_size`
- Full attention within chunks, linear across chunks
- Balances quality and memory

### Factory

```python
def build_attention(
    attention_type: str,
    embed_dim: int,
    num_heads: int,
    dropout: float,
    **kwargs
) -> Attention:
    """Build attention module from config string."""
    registry = {
        "softmax": SoftmaxAttention,
        "flash": FlashAttention,
        "linear": LinearAttention,
        "hybrid": HybridAttention,
    }
    if attention_type not in registry:
        raise ValueError(f"Unknown attention type: {attention_type}")
    return registry[attention_type](embed_dim, num_heads, dropout, **kwargs)
```

```python
class TransformerBlock(nn.Module):
    def __init__(self, config: ModelConfig):
        super().__init__()
        self.ln1 = nn.LayerNorm(config.embed_dim)
        self.attn = build_attention(
            config.attention_type,
            config.embed_dim,
            config.num_heads,
            config.dropout,
        )
        self.ln2 = nn.LayerNorm(config.embed_dim)
        self.ffn = FeedForward(config)

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.ffn(self.ln2(x))
        return x
```

## Phase 15: Linear Attention Implementation

**Commit**: `feat: implement linear attention with benchmark`

**Prerequisite**: Phase 14 complete. Attention interface exists with `build_attention()` factory.

**Modify**: `tinygpt/attention.py` only.

### Implementation

```python
class LinearAttention(Attention):
    """
    Linear Attention via kernel trick.
    Complexity: O(N) compute, O(N) memory.
    Approximate — trades quality for speed on long sequences.
    """

    def __init__(self, embed_dim: int, num_heads: int, dropout: float, **kwargs):
        super().__init__()
        self.n_heads = num_heads
        self.head_dim = embed_dim // num_heads
        self.qkv = nn.Linear(embed_dim, 3 * embed_dim, bias=False)
        self.proj = nn.Linear(embed_dim, embed_dim, bias=False)
        self.proj_drop = nn.Dropout(dropout)

        # Feature map φ(x) = elu(x) + 1 (ensures positivity)
        self.feature_map = nn.ELU()

    def forward(self, x):
        B, T, C = x.shape
        qkv = self.qkv(x).reshape(B, T, 3, self.n_heads, self.head_dim)
        q, k, v = qkv.permute(2, 0, 3, 1, 4)  # [B, H, T, D]

        # Apply feature map: φ(q), φ(k)
        q = self.feature_map(q) + 1  # [B, H, T, D] — ensures > 0
        k = self.feature_map(k) + 1  # [B, H, T, D]

        # Linear attention: φ(q) @ (φ(k).T @ v) instead of (φ(q) @ φ(k).T) @ v
        # This is O(N·D²) instead of O(N²·D)
        kv = torch.einsum("bhnd,bhne->bhde", k, v)      # [B, H, D, D]
        qkv = torch.einsum("bhnd,bhde->bhne", q, kv)    # [B, H, T, D]

        # Normalize by sum of keys (causal cumulative sum)
        k_sum = k.cumsum(dim=2)  # [B, H, T, D]
        normalizer = torch.einsum("bhnd,bhnd->bhn", q, k_sum)  # [B, H, T]
        normalizer = normalizer.unsqueeze(-1).clamp(min=1e-6)    # [B, H, T, 1]

        out = qkv / normalizer
        out = out.transpose(1, 2).reshape(B, T, C)
        return self.proj_drop(self.proj(out))

    def extra_repr(self):
        return f"n_heads={self.n_heads}, head_dim={self.head_dim}, type=linear"
```

### Benchmark Script

Create `benchmarks/attention_benchmark.py`:

```python
"""Benchmark attention implementations: memory, speed, loss."""
import torch
import time
from tinygpt.config import ModelConfig
from tinygpt.transformer import TinyGPT

def benchmark_attention(attention_type, seq_len=512, batch_size=2, n_iters=100):
    config = ModelConfig(attention_type=attention_type)
    model = TinyGPT(config).cuda().eval()
    x = torch.randint(0, 50257, (batch_size, seq_len)).cuda()

    # Warmup
    with torch.no_grad():
        for _ in range(10):
            model(x)

    # Speed benchmark
    torch.cuda.synchronize()
    start = time.perf_counter()
    with torch.no_grad():
        for _ in range(n_iters):
            model(x)
    torch.cuda.synchronize()
    elapsed = (time.perf_counter() - start) / n_iters

    # Memory benchmark
    torch.cuda.reset_peak_memory_stats()
    with torch.no_grad():
        model(x)
    peak_mem = torch.cuda.max_memory_allocated() / 1024**2

    # Param count
    params = sum(p.numel() for p in model.parameters())

    return {
        "type": attention_type,
        "seq_len": seq_len,
        "params": params,
        "time_ms": elapsed * 1000,
        "peak_mem_mb": peak_mem,
    }

if __name__ == "__main__":
    results = []
    for attn_type in ["flash", "linear"]:
        r = benchmark_attention(attn_type)
        results.append(r)
        print(f"{r['type']:>8}: {r['time_ms']:.2f} ms | {r['peak_mem_mb']:.1f} MB | {r['params']:,} params")

    # Comparison
    flash, linear = results
    speedup = flash["time_ms"] / linear["time_ms"]
    mem_ratio = flash["peak_mem_mb"] / linear["peak_mem_mb"]
    print(f"\nLinear vs Flash: {speedup:.2f}x speed, {mem_ratio:.2f}x memory")
```
